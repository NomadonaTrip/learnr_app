### Embedding Text Format

```python
def build_chunk_embedding_text(chunk: ReadingChunk, concepts: List[Concept]) -> str:
    """Build text for embedding generation."""
    concept_names = ", ".join([c.name for c in concepts])

    return f"{chunk.title}. {chunk.content} Concepts: {concept_names}"
```

**Example:**

```
Stakeholder Analysis Techniques. Stakeholder analysis is the process of analyzing
stakeholders to understand their concerns, influence, and interest in the project.
This involves identifying key stakeholders, categorizing them by influence...
Concepts: Stakeholder Analysis, Stakeholder Identification, Communication Planning
```

### Qdrant Upsert Pattern

```python
from qdrant_client import AsyncQdrantClient
from qdrant_client.models import PointStruct

async def upload_chunk_vector(
    client: AsyncQdrantClient,
    chunk_id: UUID,
    embedding: List[float],
    chunk: ReadingChunk,
    concepts: List[Concept]
):
    """Upload chunk vector to Qdrant."""
    point = PointStruct(
        id=str(chunk_id),
        vector=embedding,
        payload={
            "chunk_id": str(chunk_id),
            "course_id": str(chunk.course_id),          # NEW - required for multi-course
            "title": chunk.title,
            "knowledge_area_id": chunk.knowledge_area_id,  # CHANGED from "ka"
            "corpus_section": chunk.corpus_section,        # CHANGED from "section_ref"
            "concept_ids": [str(cid) for cid in chunk.concept_ids],
            "concept_names": [c.name for c in concepts],
            "text_content": chunk.content,
            "estimated_read_time": chunk.estimated_read_time_minutes
        }
    )

    await client.upsert(
        collection_name="reading_chunks",  # CHANGED from "babok_chunks"
        points=[point]
    )
```

### Concept Resolution

```python
async def resolve_concepts(
    concept_ids: List[UUID],
    concept_repo: ConceptRepository
) -> List[Concept]:
    """Resolve concept IDs to full concept objects."""
    # Batch fetch for efficiency
    concepts = await concept_repo.get_by_ids(concept_ids)

    # Maintain order
    concept_map = {c.id: c for c in concepts}
    return [concept_map[cid] for cid in concept_ids if cid in concept_map]
```
