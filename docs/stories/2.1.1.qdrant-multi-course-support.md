# Story 2.1.1: Qdrant Multi-Course Support

## Status

**Ready for Done**

## Story

**As a** backend developer,
**I want** to update the Qdrant collections to support multi-course architecture by adding `course_id` to payloads and renaming collections to be course-agnostic,
**so that** questions and reading chunks can be filtered by course and the system is ready for future courses beyond CBAP.

---

5. Qdrant repository updated with `course_id` filter parameter on all search methods
6. Collection initialization script updated for new schema (idempotent)
7. Existing integration tests updated to include `course_id` in payloads
8. New integration tests verify course-scoped queries work correctly
9. Health check updated to reference new collection names

### Why This Story Exists

The multi-course architecture (from `multi-course-architecture.md`) requires:

1. All content scoped to a `course_id`
2. Dynamic knowledge areas from course JSONB, not hardcoded strings
3. Questions and chunks filterable by course for isolation

Without this update:

- Story 2.2 (Concept Extraction) cannot properly scope concepts
- Story 2.4 (Question Import) cannot link questions to courses
- Story 2.5 (Question Embeddings) would embed without course context
- Future courses would require separate collection per course (poor design)

```python
# apps/api/scripts/init_qdrant_collections.py (UPDATED)
from qdrant_client import AsyncQdrantClient
from qdrant_client.models import Distance, VectorParams, PayloadSchemaType
import asyncio

async def init_collections():
    """Initialize Qdrant collections for multi-course architecture."""
    client = AsyncQdrantClient(url=settings.QDRANT_URL)

    # Collection names (course-agnostic)
    QUESTIONS_COLLECTION = "questions"
    CHUNKS_COLLECTION = "reading_chunks"

    # Get existing collections
    collections = await client.get_collections()
    collection_names = [col.name for col in collections.collections]

    # Migration: Delete old collections if they exist
    old_collections = ["cbap_questions", "babok_chunks"]
    for old_name in old_collections:
        if old_name in collection_names:
            print(f"⚠️  Deleting old collection: {old_name}")
            await client.delete_collection(collection_name=old_name)

    # Create Questions Collection
    if QUESTIONS_COLLECTION not in collection_names:
        await client.create_collection(
            collection_name=QUESTIONS_COLLECTION,
            vectors_config=VectorParams(
                size=3072,  # text-embedding-3-large
                distance=Distance.COSINE
            )
        )

        # Create payload indexes
        await client.create_payload_index(
            collection_name=QUESTIONS_COLLECTION,
            field_name="question_id",
            field_schema=PayloadSchemaType.KEYWORD
        )
        await client.create_payload_index(
            collection_name=QUESTIONS_COLLECTION,
            field_name="course_id",  # NEW - critical for multi-course
            field_schema=PayloadSchemaType.KEYWORD
        )
        await client.create_payload_index(
            collection_name=QUESTIONS_COLLECTION,
            field_name="knowledge_area_id",  # RENAMED from ka
            field_schema=PayloadSchemaType.KEYWORD
        )
        await client.create_payload_index(
            collection_name=QUESTIONS_COLLECTION,
            field_name="difficulty",
            field_schema=PayloadSchemaType.FLOAT
        )
        await client.create_payload_index(
            collection_name=QUESTIONS_COLLECTION,
            field_name="concept_ids",  # RENAMED from concept_tags
            field_schema=PayloadSchemaType.KEYWORD
        )

        print(f"✓ Created collection: {QUESTIONS_COLLECTION}")
    else:
        print(f"Collection {QUESTIONS_COLLECTION} already exists")

    # Create Reading Chunks Collection
    if CHUNKS_COLLECTION not in collection_names:
        await client.create_collection(
            collection_name=CHUNKS_COLLECTION,
            vectors_config=VectorParams(
                size=3072,
                distance=Distance.COSINE
            )
        )

        await client.create_payload_index(
            collection_name=CHUNKS_COLLECTION,
            field_name="chunk_id",
            field_schema=PayloadSchemaType.KEYWORD
        )
        await client.create_payload_index(
            collection_name=CHUNKS_COLLECTION,
            field_name="course_id",  # NEW
            field_schema=PayloadSchemaType.KEYWORD
        )
        await client.create_payload_index(
            collection_name=CHUNKS_COLLECTION,
            field_name="knowledge_area_id",  # RENAMED
            field_schema=PayloadSchemaType.KEYWORD
        )
        await client.create_payload_index(
            collection_name=CHUNKS_COLLECTION,
            field_name="section_ref",
            field_schema=PayloadSchemaType.KEYWORD
        )
        await client.create_payload_index(
            collection_name=CHUNKS_COLLECTION,
            field_name="concept_ids",  # RENAMED
            field_schema=PayloadSchemaType.KEYWORD
        )

        print(f"✓ Created collection: {CHUNKS_COLLECTION}")
    else:
        print(f"Collection {CHUNKS_COLLECTION} already exists")

    await client.close()

if __name__ == "__main__":
    asyncio.run(init_collections())
```

### Updated Repository Methods

```python
# apps/api/src/repositories/qdrant_repository.py (UPDATED)

# Collection name constants
QUESTIONS_COLLECTION = "questions"  # Was "cbap_questions"
CHUNKS_COLLECTION = "reading_chunks"  # Was "babok_chunks"

class QdrantRepository:

    async def create_question_vector(
        self,
        question_id: UUID,
        vector: List[float],
        course_id: UUID,  # NEW - required parameter
        payload: Dict[str, Any]
    ) -> None:
        """
        Insert question vector into Qdrant.

        Args:
            question_id: Question UUID
            vector: 3072-dimensional embedding vector
            course_id: Course UUID for multi-course scoping
            payload: Metadata (knowledge_area_id, difficulty, concept_ids, etc.)
        """
        # Ensure course_id is in payload
        payload["course_id"] = str(course_id)
        payload["question_id"] = str(question_id)

        point = PointStruct(
            id=str(question_id),
            vector=vector,
            payload=payload
        )

        await self.client.upsert(
            collection_name=QUESTIONS_COLLECTION,
            points=[point]
        )

    async def search_questions(
        self,
        query_vector: List[float],
        course_id: UUID = None,  # NEW - optional filter
        knowledge_area_id: str = None,  # RENAMED from ka
        difficulty_min: float = None,
        difficulty_max: float = None,
        concept_ids: List[UUID] = None,
        exclude_ids: List[UUID] = None,
        limit: int = 10
    ) -> List[Dict[str, Any]]:
        """
        Semantic search for questions with multi-course support.

        Args:
            query_vector: 3072-dimensional query embedding
            course_id: Filter by course (recommended for all queries)
            knowledge_area_id: Filter by knowledge area
            difficulty_min/max: Difficulty range filter
            concept_ids: Filter by concepts tested
            exclude_ids: Question IDs to exclude
            limit: Maximum results
        """
        must_conditions = []

        # Course filter (should almost always be set)
        if course_id:
            must_conditions.append(
                FieldCondition(
                    key="course_id",
                    match=MatchValue(value=str(course_id))
                )
            )

        # Knowledge area filter
        if knowledge_area_id:
            must_conditions.append(
                FieldCondition(
                    key="knowledge_area_id",
                    match=MatchValue(value=knowledge_area_id)
                )
            )

        # Difficulty range filter
        if difficulty_min is not None or difficulty_max is not None:
            must_conditions.append(
                FieldCondition(
                    key="difficulty",
                    range=Range(
                        gte=difficulty_min,
                        lte=difficulty_max
                    )
                )
            )

        # Concept filter (any match)
        if concept_ids:
            must_conditions.append(
                FieldCondition(
                    key="concept_ids",
                    match=MatchAny(any=[str(c) for c in concept_ids])
                )
            )

        filter_conditions = Filter(must=must_conditions) if must_conditions else None

        results = await self.client.search(
            collection_name=QUESTIONS_COLLECTION,
            query_vector=query_vector,
            query_filter=filter_conditions,
            limit=limit
        )

        # Optionally filter out excluded IDs (post-query)
        if exclude_ids:
            exclude_set = {str(eid) for eid in exclude_ids}
            results = [r for r in results if r.id not in exclude_set]

        return [
            {
                "id": result.id,
                "score": result.score,
                "payload": result.payload
            }
            for result in results
        ]

    # Similar updates for chunk methods...
```

```python
import pytest
from uuid import uuid4

# Test course ID (consistent across tests)
TEST_COURSE_ID = uuid4()
ANOTHER_COURSE_ID = uuid4()

@pytest.fixture
def sample_question_payload():
    """Sample payload with multi-course fields."""
    return {
        "question_id": str(uuid4()),
        "course_id": str(TEST_COURSE_ID),  # NEW
        "knowledge_area_id": "ba-planning",  # RENAMED (matches CBAP course JSONB)
        "difficulty": 0.5,  # Now float
        "concept_ids": [str(uuid4()), str(uuid4())],  # RENAMED
        "question_text": "What is stakeholder analysis?",
        "options": "A) ... B) ... C) ... D) ...",
        "correct_answer": "A"
    }

@pytest.mark.asyncio
async def test_search_questions_by_course_id(qdrant_repo, sample_question_vector):
    """Test that course_id filter isolates results."""
    # Create question in TEST_COURSE_ID
    q1_id = uuid4()
    await qdrant_repo.create_question_vector(
        question_id=q1_id,
        vector=sample_question_vector,
        course_id=TEST_COURSE_ID,
        payload={
            "knowledge_area_id": "ba-planning",
            "difficulty": 0.5,
            "concept_ids": [],
            "question_text": "Course 1 question"
        }
    )

    # Create question in ANOTHER_COURSE_ID
    q2_id = uuid4()
    await qdrant_repo.create_question_vector(
        question_id=q2_id,
        vector=sample_question_vector,
        course_id=ANOTHER_COURSE_ID,
        payload={
            "knowledge_area_id": "ba-planning",
            "difficulty": 0.5,
            "concept_ids": [],
            "question_text": "Course 2 question"
        }
    )

    # Search with course filter - should only find q1
    results = await qdrant_repo.search_questions(
        query_vector=sample_question_vector,
        course_id=TEST_COURSE_ID,
        limit=10
    )

    result_ids = [r["id"] for r in results]
    assert str(q1_id) in result_ids
    assert str(q2_id) not in result_ids  # Cross-course isolation

    # Cleanup
    await qdrant_repo.delete_question_vector(q1_id)
    await qdrant_repo.delete_question_vector(q2_id)
```
